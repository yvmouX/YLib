```java
package com.example.quests.menu;

/**
 * 任务列表面板 —— 完整示例。
 *
 * 这一个类覆盖了这套菜单 API 的常规用法:
 *   ① ASCII 布局 + 符号映射        ② 装饰边框与透明槽
 *   ③ 静态物品 / 动态占位符         ④ 带参数与状态判断的占位符
 *   ⑤ 分页(pageItems + perPage)     ⑥ 槽位 → 数据下标
 *   ⑦ 权限控制(visible + 点击内二次校验)
 *   ⑧ 点击后关菜单、开子菜单、原地刷新
 *   ⑨ 配置文件覆盖(generateDefault / path / version)
 *   ⑩ 只读交互(默认安全)
 *
 * ── 布局坐标表(6 行 × 9 列 = 54 格,正好是箱子菜单的上限)───────────
 *
 *      col0  1    2    3    4    5    6    7    8
 *      ┌────┬────┬────┬────┬────┬────┬────┬────┬────┐
 * row0 │ #  │ #  │ #  │ #  │ #  │ #  │ #  │ #  │ #  │  顶边框
 * row1 │ #  │ *  │ *  │ *  │ T  │ *  │ *  │ *  │ #  │  T  = 标题
 * row2 │ #  │ *  │ a  │ *  │ *  │ *  │ r  │ *  │ #  │  a  = 管理面板(需权限)  r = 刷新
 * row3 │ #  │ *  │ l  │ *  │ *  │ *  │ l  │ *  │ #  │  l  = 任务条目
 * row4 │ #  │ *  │ l  │ *  │ *  │ *  │ l  │ *  │ #  │  l  = 任务条目
 * row5 │ #  │ *  │ p  │ *  │ q  │ *  │ n  │ *  │ #  │  p/q/n = 上一页 / 页码 / 下一页
 *      └────┴────┴────┴────┴────┴────┴────┴────┴────┘
 *
 *   # = 装饰边框(不可点)   * = 透明槽(显示空气,玩家塞不进东西)
 *   l 共出现 4 次 → 每页 4 条。容量由布局决定,pageItems 不再另收一个数。
 *   每行都写满 9 个可见字符,所以列号可以直接数。
 *
 * ── 为什么不能再加一行 ────────────────────────────────────────
 *   箱子菜单上限 = InventoryType.CHEST 的 54 格 = 6 行 × 9 列。
 *   Bukkit.createInventory(owner, size) 只接受 9 的倍数且 9 <= size <= 54;
 *   即使改用更大的 Generic 容器,客户端也只画前 6 行 —— 第 7 行起玩家看不到、
 *   点不到,对箱子菜单没有意义。要放更多条目请用分页(本示例即是)。
 *
 * ── 槽位怎么数 ─────────────────────────────────────────────
 *   反引号外的每个「非空白字符」占 1 格,空格占 0 格(只用来对齐源码)。
 *   本布局刻意不靠空格凑宽度:该透明的地方写 *,该占位的地方写符号,
 *   于是每行都正好 9 格 —— 少写一个 * 就会在启动期报错,不会静默错位。
 *   多字符符号要加反引号(如 `quests`),单字符符号可裸写。
 *
 * ── 版本基线:Paper 1.20.1+ ──────────────────────────────────
 *   本示例用到的都是 1.20.1 就有的 API,可在 1.20.1 ~ 最新版之间直接跑:
 *     ✔ ItemMeta#displayName(Component) / lore(List<Component>)   (1.19 起)
 *     ✔ Material 枚举常量、Items.of(String) 走字符串解析
 *     ✘ 不用 setCustomModelDataComponent / DataComponentTypes      (1.20.5+)
 *     ✘ 不用 Enchantment.UNBREAKING 这类新常量                     (1.20.5 才改名)
 *   附魔相关一律走 Enchantment.getByKey(NamespacedKey.minecraft(...)),
 *   因为 1.20.5 把 DURABILITY / DAMAGE_ALL / DIG_SPEED 这些旧名字删掉了。
 */
@MenuConfig(
    generateDefault = true,           // 首次启动生成默认 yml;已存在则绝不覆盖
    path = "gui/questlist.yml",       // 相对插件 dataFolder;扩展名可省略
    version = 1,                      // 配置版本号,用于迁移
    sessionTtl = 1800                 // 玩家离线后保留页码等状态 1800 秒(默认值)
)
public final class QuestListMenu implements MenuTemplate {

    private final QuestService questService;

    public QuestListMenu(QuestService questService) {
        this.questService = questService;
    }

    // ══════════════════════════════════════════════════════════
    //  1) 布局
    // ══════════════════════════════════════════════════════════
    @Override
    public String[] layout() {
        return new String[]{
        //   0    1    2    3    4    5    6    7    8
            "#########",
            "#***T***#",
            "#*a***r*#",
            "#*l***l*#",
            "#*l***l*#",
            "#*p*q*n*#"
        };
    }

    @Override
    public String title() {
        return "&8任务列表";
    }

    // ══════════════════════════════════════════════════════════
    //  2) 物品与行为
    //
    //     物品一律由 .of() 提供,.ph() / .visible() / .onClick() 都是修饰。
    //     declare() 只在插件启动时跑一次,所以「按玩家不同」必须写在
    //     ph / visible / onClick 内部,不能写在这里的 if 分支里。
    // ══════════════════════════════════════════════════════════
    @Override
    public void declare(MenuBuilder b) {

        // ── 装饰边框:整块灰玻璃,点击什么都不做 ──────────────────
        // item("...") 里的字符串是 yml 的键名,服主可以改它的材质与文案
        b.symbol('#').item("frame")
            .of(Items.of(Material.GRAY_STAINED_GLASS_PANE).name(" "))
            .onClick(ctx -> { /* 故意留空:点边框没有任何反应 */ });

        // ── 透明槽:物品是空气,显示出来就是"什么都没有" ──────────
        // 但它仍是一个菜单物品 → 玩家无法把东西放进这些格子
        b.symbol('*').item("transparent")
            .of(Items.of(Material.AIR));

        // ── 标题:纯展示 ──────────────────────────────────────
        b.symbol('T').item("title")
            .of(Items.of(Material.NAME_TAG)
                     .name("&f&l任务列表")
                     .lore("&7按 &e筛选 &7分类查看"));

        // ── 管理面板按钮:无权限的玩家看到的还是原样物品,点不动 ────
        //    「显示成别的样子」要用不同符号,不能靠 ph 改物品类型,
        //    所以这里用 visible 控制显隐、用点击内二次校验兜住安全。
        b.symbol('a').item("admin_btn")
            .of(Items.of(Material.COMMAND_BLOCK)
                     .name("&c管理面板")
                     .lore("&7需要 &fquests.admin &7权限"))
            .visible(ctx -> ctx.viewer().hasPermission("quests.admin"))
            .onClick(ctx -> {
                if (!ctx.viewer().hasPermission("quests.admin")) {   // 二次校验
                    ctx.viewer().sendMessage("§c无权限");
                    return;
                }
                ctx.menu().close();
                new AdminMenu(questService).open(ctx.viewer());
            });

        // ── 刷新按钮:占位符在「每次打开/刷新」时重新求值 ──────────
        b.symbol('r').item("refresh_btn")
            .of(Items.of(Material.CLOCK)
                     .name("&f刷新")
                     .lore("&7当前共 &e{total} &7个任务", "", "&7点击原地刷新"))
            .ph("total", ctx -> String.valueOf(questService.count(ctx.viewer())))
            .onClick(ctx -> ctx.menu().refresh());   // 原地重渲染,不重建背包

        // ── 任务列表:分页 ─────────────────────────────────────
        //    l 在布局里出现 4 次 → 每页 4 条。这个数由框架算好交进来,
        //    代码里不再声明,也就没有"改了布局忘了改常量"这回事。
        b.symbol('l').item("quest_entry")
            .pageItems(new PageSource() {
                @Override
                public List<ItemStack> fetch(Player viewer, int page, int perPage) {
                    List<ItemStack> out = new ArrayList<>();
                    for (Quest q : questService.page(viewer, page, perPage)) {
                        out.add(Items.of(q.done() ? Material.LIME_DYE : Material.GRAY_DYE)
                                       .name((q.done() ? "&a" : "&f") + q.name())
                                       .lore("&7进度: &e" + q.progressText(),
                                             "&7奖励: &6" + q.rewardText(),
                                             "",
                                             q.done() ? "&a已完成" : "&7点击查看详情")
                                       .build());
                    }
                    return out;
                    // 末页不足 perPage 条时,框架把多出来的槽位【显式写成空气】——
                    // 不是"跳过",所以上一页的任务绝不会残留下来被误点。
                    // 超过 perPage 条则截断并打 warning(说明没按 perPage 切片)。
                }

                @Override
                public int total(Player viewer) {
                    return questService.count(viewer);   // 供 pageCount() 计算总页数
                }
            })
            .onClick((slot, ctx) -> {
                // 槽位 → 页内下标 → 全局下标
                int global = ctx.page() * ctx.perPage() + ctx.indexInSymbol(slot);
                Quest q = questService.at(ctx.viewer(), global);
                if (q != null) {
                    new QuestDetailMenu(questService, q).open(ctx.viewer());
                }
            });

        // ── 上一页:边界判断在业务侧(框架不知道总条数)──────────
        b.symbol('p').item("prev_btn")
            .of(Items.of(Material.ARROW).name("&f上一页"))
            .onClick(ctx -> {
                if (ctx.page() > 0) {
                    ctx.page(ctx.page() - 1);
                    ctx.menu().refresh();
                }
            });

        // ── 页码指示器:整页共享一个值 → 用 ph(不是 pageItems)────
        b.symbol('q').item("page_info")
            .of(Items.of(Material.PAPER).name("&f第 &e{page}&7/&e{pages} &f页"))
            .ph("page",  ctx -> String.valueOf(ctx.page() + 1))                        // 显示用 1-based
            .ph("pages", ctx -> ctx.pageCount() < 0 ? "?" : String.valueOf(ctx.pageCount()));

        // ── 下一页:用 pageCount() 判断,不要靠"返回空列表"猜到底 ──
        b.symbol('n').item("next_btn")
            .of(Items.of(Material.ARROW).name("&f下一页"))
            .onClick(ctx -> {
                int next = ctx.page() + 1;
                if (next < ctx.pageCount()) {
                    ctx.page(next);
                    ctx.menu().refresh();
                }
            });

        // ── 兜底填充:布局里没有被任何符号占用的格子 ─────────────
        //    本布局每格都有归属,所以这行实际没有生效对象;
        //    留着是为了说明:删掉一行 # 边框时,空出来的格子会由它补上。
        b.fillEmpty(Items.of(Material.BLACK_STAINED_GLASS_PANE).name(" "));

        // ── 交互策略:只读 ────────────────────────────────────
        //    玩家不能拖走/塞入任何物品;点没有 handler 的格子静默取消。
        //    框架负责拦:InventoryClickEvent / InventoryDragEvent /
        //    InventoryMoveItemEvent / InventoryPickupItemEvent,
        //    以及 MOVE_TO_OTHER_INVENTORY、COLLECT_TO_CURSOR、
        //    HOTBAR_SWAP、DROP_* 这些 InventoryAction。
        b.interaction(Interaction.READ_ONLY);

        // 允许移动的容器(仓库类)改成下面这样,并且按需设置 onOpen / onClose:
        //   b.interaction(Interaction.ALLOW_MOVE);
        //   b.delayedRefresh(false);        // ALLOW_MOVE 时必须 false,否则打断拖拽
    }

    // ══════════════════════════════════════════════════════════
    //  3) 分页状态说明
    //
    //     MenuSession 的维度是「玩家 × 菜单」,不在全局共享的 MenuDefinition 里,
    //     并且 close() 时【不销毁】:
    //        玩家点进详情 → 返回列表 → 还停在第 3 页
    //     想让"重新打开就回到第一页",显式调用:
    //        menu.resetSession(player);
    //
    //     put(key, value) 只能放轻量值(页码、枚举、字符串)。
    //     往 session 里塞 List<ItemStack> 或数据库结果 = 把玩家数据常驻内存。
    // ══════════════════════════════════════════════════════════
}
```
